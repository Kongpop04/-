print("=== โปรแกรมคำนวณพื้นที่รูปทรง ===")
print("1. สี่เหลี่ยมมุมฉาก")
print("2. สามเหลี่ยม")
print("3. วงกลม")

เลือก = input("เลือกรูปทรง (1-3): ")

if เลือก == "1":
    กว้าง = float(input("กรอกความกว้าง: "))
    ยาว = float(input("กรอกความยาว: "))
    พื้นที่ = กว้าง * ยาว
    print("พื้นที่สี่เหลี่ยมมุมฉาก =", พื้นที่)

elif เลือก == "2":
    ฐาน = float(input("กรอกความยาวฐาน: "))
    สูง = float(input("กรอกความสูง: "))
    พื้นที่ = 0.5 * ฐาน * สูง
    print("พื้นที่สามเหลี่ยม =", พื้นที่)

elif เลือก == "3":
    รัศมี = float(input("กรอกรัศมี: "))
    พื้นที่ = 3.14159 * รัศมี ** 2
    print("พื้นที่วงกลม =", พื้นที่)

else:
    print("กรุณาเลือกหมายเลข 1-3")
